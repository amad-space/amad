    <script>
        const sampleMessages = [
            { sender: "إلى غريب يبحث عن طمأنينة", text: "ستمر الأيام العسرة وتزهر الأماكن التي ظننت أنها لن تروى أبداً. اطمئن." },
            { sender: "إلى نفسي بعد عامين", text: "أتمنى أنك وصلت للمكان الذي كنت تحارب من أجله وأنت تبتسم." },
            { sender: "أثر طيب", text: "الكلمة الطيبة سراج يمشي في صدر صاحبه، دعه يضيء عتمتك اليوم." }
        ];

        const skyCanvas = document.getElementById('skyCanvas');
        
        for (let i = 0; i < 35; i++) {
            createStarNode(
                Math.random() * window.innerWidth,
                Math.random() * window.innerHeight,
                sampleMessages[Math.floor(Math.random() * sampleMessages.length)]
            );
        }

        function createStarNode(x, y, messageData) {
            const star = document.createElement('div');
            star.classList.add('star-node');
            const size = (Math.random() * 4 + 3) + 'px';
            star.style.width = size;
            star.style.height = size;
            star.style.left = x + 'px';
            star.style.top = y + 'px';
            star.style.animationDuration = (Math.random() * 3 + 2) + 's';

            star.onclick = () => {
                document.getElementById('modalTitle').innerText = messageData.sender;
                document.getElementById('modalText').innerText = `"${messageData.text}"`;
                document.getElementById('starModal').style.display = 'flex';
            };

            skyCanvas.appendChild(star);
        }

        function closeModal() {
            document.getElementById('starModal').style.display = 'none';
        }

        function setTheme(bgColor, accentColor, accentHover, borderColor) {
            document.documentElement.style.setProperty('--bg-color', bgColor);
            document.documentElement.style.setProperty('--accent-color', accentColor);
            document.documentElement.style.setProperty('--accent-hover', accentHover);
            document.documentElement.style.setProperty('--border-color', borderColor);
        }

        function switchTab(tabId, btn) {
            document.querySelectorAll('.section-content').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
            document.getElementById(tabId).classList.add('active');
            btn.classList.add('active');
        }

        document.getElementById('timeCapsuleForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const recipient = document.getElementById('recipientName').value;
            const content = document.getElementById('messageContent').value;

            createStarNode(
                Math.random() * (window.innerWidth - 100) + 50,
                Math.random() * (window.innerHeight - 100) + 50,
                { sender: recipient, text: content }
            );

            document.getElementById('timeCapsuleForm').style.display = 'none';
            document.getElementById('successBox').style.display = 'block';
        });

        function orderCard(cardName) {
            const text = encodeURIComponent(`مرحباً، أرغب في طلب (${cardName}) بمبلغ 500 فلس من موقع أَمَد ✨`);
            window.open(`https://api.whatsapp.com/send?phone=&text=${text}`, '_blank');
        }
